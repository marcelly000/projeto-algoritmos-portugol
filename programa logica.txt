programa
{
  funcao inicio()
    {
      inteiro v[8]
      inteiro i
      inteiro soma = 0
      para (i = 0; i < 8; i++)
        {
          escreva("Digite o ", i + 1, " numero: ")
          leia(v[i])
          soma = soma + v[i]
        }
        escreva("\nDobro dos numeros:\n")
        para (i = 0; i < 8; i++)
        {
            escreva(v[i] * 2, "\n")
        }
        escreva("\nSoma de todos: ", soma)
    }
}
